   121→        fun isPrefetchApplicable(pages: List<ReaderPage>): Boolean {
   122→                val repo = repository
   123→                val hasLocalPages = repo?.source == LocalMangaSource || pages.any { it.isLocalPage() }
   124→                val canPrefetchSource = repo is CachingMangaRepository || hasLocalPages
   125→                return canPrefetchSource
   126→                        && (hasLocalPages || settings.isPagesPreloadEnabled)
   127→                        && !context.isPowerSaveMode()
   128→                        && !isLowRam()
   129→        }

   131→        @AnyThread
   132→        fun prefetch(pages: List<ReaderPage>) = loaderScope.launch {
   133→                var hasPendingPrefetch = false
   134→                prefetchLock.withLock {
   135→                        val queueLimit = getPrefetchQueueLimit()
   136→                        if (queueLimit <= 0) {
   137→                                prefetchQueue.clear()
   138→                                return@withLock
   139→                        }
   140→                        for (page in pages.asReversed()) {
   141→                                val key = page.toMangaPage().taskKey()
   142→                                val hasTask = synchronized(tasks) {
   143→                                        tasks.containsKey(key)
   144→                                }
   145→                                if (hasTask) {
   146→                                        continue
   147→                                }
   148→                                prefetchQueue.offerFirst(page.toMangaPage())
   149→                                if (prefetchQueue.size > queueLimit) {
   150→                                        prefetchQueue.pollLast()
   151→                                }
   152→                        }
   153→                        hasPendingPrefetch = prefetchQueue.isNotEmpty()
   154→                }
   155→                if (hasPendingPrefetch && counter.get() == 0) {
   156→                        onIdle()
   157→                }
   158→        }

   160→        suspend fun loadPreview(page: MangaPage): ImageSource? {
   161→                val preview = page.preview
   162→                if (preview.isNullOrEmpty()) {
   163→                        return null
   164→                }
   165→                val request = ImageRequest.Builder(context)
   166→                        .data(preview)
   167→                        .mangaSourceExtra(page.source)
   168→                        .transformations(TrimTransformation())

   173→        fun peekPreviewSource(preview: String?): ImageSource? {
   174→                if (preview.isNullOrEmpty()) {
   175→                        return null
   176→                }
   177→                coil.memoryCache?.let { cache ->
   178→                        val key = MemoryCache.Key(preview)
   179→                        cache[key]?.image?.let {
   180→                                return if (it is BitmapImage) {
   181→                                        ImageSource.cachedBitmap(it.toBitmap())
   182→                                } else {
   183→                                        ImageSource.bitmap(it.toBitmap())
   184→                                }
   185→                        }
   186→                }
   187→                coil.diskCache?.let { cache ->
   188→                        cache.openSnapshot(preview)?.use { snapshot ->
   189→                                return ImageSource.file(snapshot.data.toFile())
   190→                        }
   191→                }
   192→                return null
   193→        }

   195→        fun loadPageAsync(page: MangaPage, force: Boolean): ProgressDeferred<Uri, Float> {
   196→                val key = page.taskKey()
   197→                var task = synchronized(tasks) {
   198→                        tasks[key]
   199→                }?.takeIf { it.isValid() }
   200→                if (force) {
   201→                        task?.cancel()
   202→                } else if (task?.isCancelled == false) {
   203→                        return task
   204→                }
   205→                task = loadPageAsyncImpl(page, skipCache = force, isPrefetch = false)
   206→                synchronized(tasks) {
   207→                        tasks[key] = task
   208→                }
   209→                return task
   210→        }

   216→        @CheckResult
   217→        suspend fun materializeZipEntry(uri: Uri): Uri {
   218→                check(uri.isZipUri()) { "Expected ZIP uri: $uri" }
   219→                val cacheKey = rawZipCacheKey(uri)
   220→                cache.get(cacheKey)?.let { file ->
   221→                        if (file.isUsablePageCache()) {
   222→                                return file.toUri()
   223→                        }
   224→                        cache.remove(cacheKey)
   225→                }
   226→                return withContext(Dispatchers.IO) {
   227→                        ZipFile(uri.schemeSpecificPart).use { zip ->
   228→                                val entryName = uri.fragment ?: throw FileNotFoundException(uri.toString())
   229→                                val entry = zip.getEntry(entryName) ?: throw FileNotFoundException("$uri#$entryName")
   230→                                zip.getInputStream(entry).source().use { source ->
   231→                                        val file = cache.set(cacheKey, source, MimeTypes.getMimeTypeFromExtension(entry.name))
   232→                                        if (file.isUsablePageCache()) {
   233→                                                file.toUri()
   234→                                        } else {
   235→                                                cache.remove(cacheKey)
   236→                                                throw UnsupportedFileException("Unsupported or corrupt image file")
   237→                                        }
   238→                                }
   239→                        }
   240→                }
   241→        }

   243→        @CheckResult
   244→        suspend fun convertBimap(uri: Uri): Uri = convertLock.withLock {
   245→                if (uri.isZipUri()) {
   246→                        val cacheKey = convertedZipCacheKey(uri)
   247→                        cache.get(cacheKey)?.let { file ->
   248→                                if (file.isUsablePageCache()) {
   249→                                        return@withLock file.toUri()
   250→                                }
   251→                                cache.remove(cacheKey)
   252→                        }
   253→                        runInterruptible(Dispatchers.IO) {
   254→                                ZipFile(uri.schemeSpecificPart).use { zip ->
   255→                                        val entryName = uri.fragment ?: throw FileNotFoundException(uri.toString())
   256→                                        val entry = zip.getEntry(entryName) ?: throw FileNotFoundException("$uri#$entryName")
   257→                                        context.ensureRamAtLeast(estimateDecodeBytes(entry.size.coerceAtLeast(0L)))
   258→                                        zip.getInputStream(entry).use {
   259→                                                BitmapDecoderCompat.decode(it, MimeTypes.getMimeTypeFromExtension(entry.name))
   260→                                        }
   261→                                }
   262→                        }.use { image ->
   263→                                cache.set(cacheKey, image).toUri()
   264→                        }
   265→                } else {
   266→                        // For a plain file URI, write the converted PNG to a NEW cache
   267→                        // entry (not back to the source) so the original is preserved and
   268→                        // the conversion works even when the source lives in read-only
   269→                        // storage (e.g. SAF tree the user only granted read access to).
   270→                        val cacheKey = convertedFileCacheKey(uri)
   271→                        cache.get(cacheKey)?.let { file ->
   272→                                if (file.isUsablePageCache()) {
   273→                                        return@withLock file.toUri()
   274→                                }
   275→                                cache.remove(cacheKey)
   276→                        }
   277→                        val source = uri.toFile()
   278→                        runInterruptible(Dispatchers.IO) {
   279→                                context.ensureRamAtLeast(source.estimateDecodeMemoryBytes())
   280→                                BitmapDecoderCompat.decode(source)
   281→                        }.use { image ->
   282→                                cache.set(cacheKey, image).toUri()
   283→                        }
   284→                }
   285→        }

   315→        private fun onIdle() = loaderScope.launch {
   316→                prefetchLock.withLock {
   317→                        while (prefetchQueue.isNotEmpty()) {
   318→                                val page = prefetchQueue.pollFirst() ?: return@launch
   319→                                synchronized(tasks) {
   320→                                        tasks[page.taskKey()] = loadPageAsyncImpl(page, skipCache = false, isPrefetch = true)
   321→                                }
   322→                        }
   323→                }
   324→        }
   325→
   326→        private fun loadPageAsyncImpl(
   327→                page: MangaPage,
   328→                skipCache: Boolean,
   329→                isPrefetch: Boolean,
   330→        ): ProgressDeferred<Uri, Float> {
   331→                val progress = MutableStateFlow(PROGRESS_UNDEFINED)
   332→                val deferred = loaderScope.async {
   333→                        counter.incrementAndGet()
   334→                        try {
   335→                                loadPageImpl(
   336→                                        page = page,
   337→                                        progress = progress,
   338→                                        isPrefetch = isPrefetch,
   339→                                        skipCache = skipCache,
   340→                                )
   341→                        } finally {
   342→                                if (counter.decrementAndGet() == 0) {
   343→                                        onIdle()
   344→                                }
   345→                        }
   346→                }
   347→                return ProgressDeferred(deferred, progress)
   348→        }

   350→        @Synchronized

   352→                val result = repository
   353→                return if (result != null && result.source == source) {
   354→                        result
   355→                } else {
   356→                        mangaRepositoryFactory.create(source).also { repository = it }
   357→                }

   360→        private suspend fun loadPageImpl(
   361→                page: MangaPage,
   362→                progress: MutableStateFlow<Float>,
   363→                isPrefetch: Boolean,
   364→                skipCache: Boolean,
   365→        ): Uri = semaphore.withPermit {
   366→                val pageUrl = resolvePageUrl(page)
   367→                check(pageUrl.isNotBlank()) { "Cannot obtain full image url for $page" }
   368→                if (!skipCache) {
   369→                        cache.get(pageUrl)?.let { file ->
   370→                                if (file.isUsablePageCache()) {
   371→                                        recordLoadedPageFootprint(file)
   372→                                        return file.toUri()
   373→                                }
   374→                                cache.remove(pageUrl)
   375→                        }
   376→                }
   377→                val uri = pageUrl.toUri()
   378→                return when {
   379→                        uri.isZipUri() -> {
   380→                                val normalizedUri = if (uri.scheme == URI_SCHEME_ZIP) {
   381→                                        uri
   382→                                } else { // legacy uri
   383→                                        uri.buildUpon().scheme(URI_SCHEME_ZIP).build()
   384→                                }
   385→                                materializeZipEntry(normalizedUri).also {
   386→                                        recordLoadedPageFootprint(it.toFile())
   387→                                }
   388→                        }
   389→
   390→                        uri.isFileUri() -> uri.also {
   391→                                if (!it.toFile().isUsablePageCache()) {
   392→                                        throw UnsupportedFileException("Unsupported or corrupt image file")
   393→                                }
   394→                                recordLoadedPageFootprint(it.toFile())
   395→                        }
   396→                        else -> {
   397→                                if (isPrefetch) {
   398→                                        val slowdownDelayMs = downloadSlowdownDispatcher.getDelayMs(page.source)
   399→                                        if (slowdownDelayMs > 0L) {
   400→                                                delay(slowdownDelayMs)
   401→                                        }
   402→                                }
   403→                                val request = createPageRequest(pageUrl, page.source)
   404→                                imageProxyInterceptor.interceptPageRequest(request, okHttp).ensureSuccess().use { response ->
   405→                                        response.requireBody().withProgress(progress).use {
   406→                                                cache.set(pageUrl, it.source(), it.contentType()?.toMimeType())
   407→                                        }
   408→                                }.also { file ->
   409→                                        if (!file.isUsablePageCache()) {
   410→                                                cache.remove(pageUrl)
   411→                                                throw UnsupportedFileException("Unsupported or corrupt image file")
   412→                                        }
   413→                                        recordLoadedPageFootprint(file)
   414→                                }.toUri()
   415→                        }
   416→                }
   417→        }

   437→        private fun getPrefetchQueueLimit(): Int {
   438→                val availableRam = context.ramAvailable
   439→                if (context.isPowerSaveMode() || isLowRam(availableRam)) {
   440→                        return 0
   441→                }
   442→                val baseLimit = when {
   443→                        availableRam >= FileSize.MEGABYTES.convert(PREFETCH_HIGH_RAM_MB, FileSize.BYTES) -> PREFETCH_LIMIT_HIGH
   444→                        availableRam >= FileSize.MEGABYTES.convert(PREFETCH_DEFAULT_RAM_MB, FileSize.BYTES) -> PREFETCH_LIMIT_DEFAULT
   445→                        else -> PREFETCH_LIMIT_LOW
   446→                }
   447→                return when {
   448→                        lastPageDecodeBytes >= megabytes(PREFETCH_HUGE_PAGE_MB) -> minOf(baseLimit, PREFETCH_LIMIT_HUGE_PAGE)
   449→                        lastPageDecodeBytes >= megabytes(PREFETCH_LARGE_PAGE_MB) -> minOf(baseLimit, PREFETCH_LIMIT_LARGE_PAGE)
   450→                        else -> baseLimit
   451→                }
   452→        }
   453→
   454→        private fun isLowRam(availableRam: Long): Boolean {
   455→                return availableRam <= FileSize.MEGABYTES.convert(PREFETCH_MIN_RAM_MB, FileSize.BYTES)
   456→        }

   458→        private fun recordLoadedPageFootprint(file: File) {
   459→                val estimate = file.estimateDecodeMemoryBytes()
   460→                if (estimate > 0L) {
   461→                        lastPageDecodeBytes = estimate
   462→                }
   463→        }
   464→
   465→        private fun File.isUsablePageCache(): Boolean {
   466→                return runCatchingCancellable {
   467→                        if (!isFile || length() == 0L) {
   468→                                return@runCatchingCancellable false
   469→                        }
   470→                        val nativeInfo = if (nativeImageProbe.isAvailable) {
   471→                                nativeImageProbe.probe(this)
   472→                        } else {
   473→                                null
   474→                        }
   475→                        if (nativeInfo?.isCorrupt == true) {
   476→                                return@runCatchingCancellable false
   477→                        }
   478→                        val detectedType = nativeInfo?.mimeType?.toMimeTypeOrNull()
   479→                                ?: MimeTypes.probeMimeType(this)
   480→                                ?: BitmapDecoderCompat.probeMimeType(this)
   481→                        detectedType?.isImage == true
   482→                }.getOrDefault(false)
   483→        }
   484→
   485→        private fun File.estimateDecodeMemoryBytes(): Long {
   486→                val nativeEstimate = if (nativeImageProbe.isAvailable) {
   487→                        nativeImageProbe.probe(this)?.estimatedMemoryBytes
   488→                } else {
   489→                        null
   490→                }
   491→                return nativeEstimate?.takeIf { it > 0L } ?: estimateDecodeBytes(length())
   492→        }
   493→
   494→        private fun estimateDecodeBytes(encodedBytes: Long): Long {
   495→                val normalizedBytes = encodedBytes.coerceAtLeast(0L)
   496→                val scaledBytes = if (normalizedBytes > Long.MAX_VALUE / IMAGE_DECODE_FALLBACK_MULTIPLIER) {
   497→                        Long.MAX_VALUE
   498→                } else {
   499→                        normalizedBytes * IMAGE_DECODE_FALLBACK_MULTIPLIER
   500→                }
   501→                return scaledBytes
   502→                        .coerceAtLeast(megabytes(IMAGE_DECODE_MIN_MB))
   503→                        .coerceAtMost(megabytes(IMAGE_DECODE_MAX_MB))
   504→        }

   538→        companion object {

   540→                private const val PROGRESS_UNDEFINED = -1f
   541→                private const val PREFETCH_LIMIT_LOW = 3
   542→                private const val PREFETCH_LIMIT_DEFAULT = 6
   543→                private const val PREFETCH_LIMIT_HIGH = 10
   544→                private const val PREFETCH_MIN_RAM_MB = 80L
   545→                private const val PREFETCH_DEFAULT_RAM_MB = 160L
   546→                private const val PREFETCH_HIGH_RAM_MB = 512L
   547→                private const val PREFETCH_LARGE_PAGE_MB = 32L
   548→                private const val PREFETCH_HUGE_PAGE_MB = 64L
   549→                private const val PREFETCH_LIMIT_LARGE_PAGE = 2

   574→                @Blocking
   575→                private fun Uri.exists(): Boolean = when {
   576→                        isFileUri() -> toFile().exists()
   577→                        // Do not open the ZIP here: this runs on every loadPageAsync() call
   578→                        // (often on the main thread) and parsing the central directory of a
   579→                        // large CBZ archive per page causes jank while reading downloaded
   580→                        // manga. A stale/missing entry is handled by the load failing and
   581→                        // being retried with a fresh task.
   582→                        isZipUri() -> File(requireNotNull(schemeSpecificPart)).exists()
   583→
   584→                        else -> false
   585→                }
   586→
   587→                @Blocking
   588→                private fun Uri.isTargetNotEmpty(): Boolean = when {
   589→                        isFileUri() -> toFile().isNotEmpty()
   590→                        isZipUri() -> File(requireNotNull(schemeSpecificPart)).isNotEmpty()
   591→
   592→                        else -> false
   593→                }
   594→        }
