   338→   338→    @MainThread
   339→   339→    fun onCurrentPageChanged(lowerPos: Int, upperPos: Int) {
   340→   340→        val prevJob = stateChangeJob
   341→   341→        val pages = content.value.pages // capture immediately
   342→   342→        val centerPos = (lowerPos + upperPos) / 2
   343→   343→        val capturedPage = pages.getOrNull(centerPos)
   344→   344→        // Capture the visible window relative to the captured center, so the
   345→   345→        // relative position around the page is preserved even when the page
   346→   346→        // list changes (chapter switch) between the main-thread capture and
   347→   347→        // the post-join recompute below.
   348→   348→        val visibleBeforeCapture = (centerPos - lowerPos).coerceAtLeast(0)
   349→   349→        val visibleAfterCapture = (upperPos - centerPos).coerceAtLeast(0)
   350→   350→        stateChangeJob = launchJob(Dispatchers.Default) {
   351→   351→            prevJob?.cancelAndJoin()
   352→   352→            loadingJob?.join()
   353→   353→            val currentPages = content.value.pages
   354→   354→            val resolvedCenterPos = capturedPage?.let { page ->
   355→   355→                currentPages.indexOfFirst {
   356→   356→                    it.id == page.id && it.chapterId == page.chapterId && it.index == page.index
   357→   357→                }.takeIf { it >= 0 }
   358→   358→            } ?: centerPos.coerceIn(0, currentPages.lastIndex.coerceAtLeast(0))
   359→   359→            val currentPage = currentPages.getOrNull(resolvedCenterPos)
   360→   360→            if (currentPage != null) {
   361→   361→                readingState.update { cs ->
   362→   362→                    cs?.copy(chapterId = currentPage.chapterId, page = currentPage.index)
   363→   363→                }
   364→   364→            }
   365→   365→            notifyStateChanged()
   366→   366→            if (currentPages.isEmpty() || loadingJob?.isActive == true) {
   367→   367→                return@launchJob
   368→   368→            }
   369→   369→            ensureActive()
   370→   370→            // Clamp the captured window to the post-join page list size so we
   371→   371→            // never try to read or prefetch past the end after a chapter shrink.
   372→   372→            val lastIndex = currentPages.lastIndex
   373→   373→            val visibleBefore = visibleBeforeCapture.coerceAtMost(resolvedCenterPos)
   374→   374→            val visibleAfter = visibleAfterCapture.coerceAtMost(lastIndex - resolvedCenterPos)
   375→   375→            val resolvedLowerPos = (resolvedCenterPos - visibleBefore).coerceAtLeast(0)
   376→   376→            val resolvedUpperPos = (resolvedCenterPos + visibleAfter).coerceAtMost(lastIndex)
   377→   377→            val autoLoadAllowed = readerMode.value != ReaderMode.WEBTOON || !isWebtoonPullGestureEnabled.value
   378→   378→            if (autoLoadAllowed) {
   379→   379→                if (resolvedUpperPos >= currentPages.lastIndex - BOUNDS_PAGE_OFFSET) {
   380→   380→                    loadPrevNextChapter(currentPages.last().chapterId, isNext = true)
   381→   381→                }
   382→   382→                if (resolvedLowerPos <= BOUNDS_PAGE_OFFSET) {
   383→   383→                    loadPrevNextChapter(currentPages.first().chapterId, isNext = false)
   384→   384→                }
   385→   385→            }
   386→   386→            val prefetchPages = currentPages.trySublist(
   387→   387→                fromIndex = resolvedUpperPos + 1,
   388→   388→                toIndex = resolvedUpperPos + 1 + PREFETCH_LIMIT,
   389→   389→            )
   390→   390→            if (pageLoader.isPrefetchApplicable(prefetchPages)) {
   391→   391→                pageLoader.prefetch(prefetchPages)
   392→   392→            }
   393→   393→        }
   394→   394→    }

   432→   432→    private fun loadImpl() {
   433→   433→        loadingJob = launchLoadingJob(Dispatchers.Default + EventExceptionHandler(onLoadingError)) {
   434→   434→            var exception: Exception? = null
   435→   435→            var loadedDetails: MangaDetails? = null
   436→   436→            try {
   437→   437→                detailsLoadUseCase(intent, force = false, preferLocalChapters = true)
   438→   438→                    .collect { details ->
   439→   439→                        loadedDetails = details
   440→   440→                        if (mangaDetails.value == null) {
   441→   441→                            mangaDetails.value = details
   442→   442→                        }
   443→   443→                        chaptersLoader.init(details)
   444→   444→                        val manga = details.toManga()
   445→   445→                        // obtain state
   446→   446→                        if (readingState.value == null) {
   447→   447→                            val newState = getStateFromIntent(manga)
   448→   448→                            if (newState == null) {
   449→   449→                                return@collect // manga not loaded yet if cannot get state
   450→   450→                            }
   451→   451→                            if (!details.canLoadInitialChapter(newState.chapterId)) {
   452→   452→                                mangaDetails.value = details.filterChapters(selectedBranch.value)
   453→   453→                                return@collect
   454→   454→                            }
   455→   455→                            readingState.value = newState
   456→   456→                            val mode = runCatchingCancellable {
   457→   457→                                detectReaderModeUseCase(manga, newState)
   458→   458→                            }.getOrDefault(settings.defaultReaderMode)
   459→   459→                            val branch = chaptersLoader.peekChapter(newState.chapterId)?.branch
   460→   460→                            selectedBranch.value = branch
   461→   461→                            readerMode.value = mode
   462→   462→                            try {
   463→   463→                                chaptersLoader.loadSingleChapter(newState.chapterId)
   464→   464→                                preloadNextChapterAfterLocal(newState.chapterId, details)
   465→   465→                            } catch (e: Exception) {
   466→   466→                                readingState.value = null // try next time
   467→   467→                                exception = e.mergeWith(exception)
   468→   468→                                return@collect
   469→   469→                            }
   470→   470→                        }
   471→   471→                        mangaDetails.value = details.filterChapters(selectedBranch.value)
   472→   472→
   473→   473→                        // save state
   474→   474→                        if (!isIncognitoMode.firstNotNull()) {
   475→   475→                            readingState.value?.let {
   476→   476→                                val percent = computePercent(it.chapterId, it.page)
   477→   477→                                historyUpdateUseCase(manga, it, percent)
   478→   478→                            }
   479→   479→                        }
   480→   480→                        notifyStateChanged()
   481→   481→                        content.value = ReaderContent(chaptersLoader.snapshot(), readingState.value)
   482→   482→                    }
   483→   483→            } catch (e: CancellationException) {
   484→   484→                throw e
   485→   485→            } catch (e: Exception) {
   486→   486→                exception = e.mergeWith(exception)
   487→   487→            }
   488→   488→            if (readingState.value == null) {
   489→   489→                val loadedManga = loadedDetails // for smart cast
   490→   490→                if (loadedManga != null) {
   491→   491→                    mangaDetails.value = loadedManga.filterChapters(selectedBranch.value)
   492→   492→                }
   493→   493→                val loadingError = when {
   494→   494→                    exception != null -> exception
   495→   495→                    loadedManga == null || !loadedManga.isLoaded -> null
   496→   496→                    loadedManga.isRestricted -> EmptyMangaException(
   497→   497→                        EmptyMangaReason.RESTRICTED,
   498→   498→                        loadedManga.toManga(),
   499→   499→                        null,
   500→   500→                    )
   501→   501→
   502→   502→                    loadedManga.allChapters.isEmpty() -> EmptyMangaException(
   503→   503→                        EmptyMangaReason.NO_CHAPTERS,
   504→   504→                        loadedManga.toManga(),
   505→   505→                        null,
   506→   506→                    )
   507→   507→
   508→   508→                    else -> null
   509→   509→                } ?: IllegalStateException("Unable to load manga. This should never happen. Please report")
   510→   510→                onLoadingError.call(loadingError)
   511→   511→            } else exception?.let { e ->
   512→   512→                // manga has been loaded but error occurred
   513→   513→                errorEvent.call(e)
   514→   514→            }
   515→   515→        }
   516→   516→    }

   522→   522→    private suspend fun preloadNextChapterAfterLocal(currentId: Long, details: MangaDetails) {
   523→   523→        if (!settings.isPagesPreloadEnabled) {
   524→   524→            return
   525→   525→        }
   526→   526→        if (readerMode.value == ReaderMode.WEBTOON && isWebtoonPullGestureEnabled.value) {
   527→   527→            return
   528→   528→        }
   529→   529→        val chapter = chaptersLoader.peekChapter(currentId) ?: return
   530→   530→        if (chapter.source != LocalMangaSource) {
   531→   531→            return
   532→   532→        }
   533→   533→        runCatchingCancellable {
   534→   534→            chaptersLoader.loadPrevNextChapter(details, currentId, isNext = true)

   548→   548→    private fun <T> List<T>.trySublist(fromIndex: Int, toIndex: Int): List<T> {
   549→   549→        val fromIndexBounded = fromIndex.coerceIn(0, size)
   550→   550→        val toIndexBounded = toIndex.coerceIn(fromIndexBounded, size)
   551→   551→        return if (fromIndexBounded == toIndexBounded) {
   552→   552→            emptyList()
   553→   553→        } else {
   554→   554→            subList(fromIndexBounded, toIndexBounded)
   555→   555→        }
   556→   556→    }

   570→   570→            currentPage = state.page,

   572→   572→            incognito = isIncognitoMode.value == true,
   573→   573→        )
   574→   574→        uiState.value = newState
   575→   575→        if (isIncognitoMode.value == false) {
   576→   576→            statsCollector.onStateChanged(m.id, state)
   577→   577→            discordRpc.updateRpc(m.toManga(), newState)
   578→   578→        }
   579→   579→    }

   581→   581→    private fun computePercent(chapterId: Long, pageIndex: Int): Float {
   582→   582→        val branch = chaptersLoader.peekChapter(chapterId)?.branch
   583→   583→        val chapters = mangaDetails.value?.chapters?.get(branch) ?: return PROGRESS_NONE
   584→   584→        val chaptersCount = chapters.size
   585→   585→        val chapterIndex = chapters.indexOfFirst { x -> x.id == chapterId }
   586→   586→        val pagesCount = chaptersLoader.getPagesCount(chapterId)
   587→   587→        if (chaptersCount == 0 || pagesCount == 0) {
   588→   588→            return PROGRESS_NONE
   589→   589→        }
   590→   590→        val pagePercent = (pageIndex + 1) / pagesCount.toFloat()
   591→   591→        val ppc = 1f / chaptersCount
   592→   592→        return ppc * chapterIndex + ppc * pagePercent
   593→   593→    }

   630→   630→    private suspend fun getStateFromIntent(manga: Manga): ReaderState? {
   631→   631→        // check if we have at least some chapters loaded
   632→   632→        if (manga.chapters.isNullOrEmpty()) {
   633→   633→            return null
   634→   634→        }
   635→   635→        // specific state is requested
   636→   636→        val requestedState: ReaderState? = savedStateHandle[ReaderIntent.EXTRA_STATE]
   637→   637→        if (requestedState != null) {
   638→   638→            return if (manga.findChapterById(requestedState.chapterId) != null) {
   639→   639→                requestedState
   640→   640→            } else {
   641→   641→                null
   642→   642→            }
   643→   643→        }
   644→   644→
   645→   645→        val requestedBranch: String? = savedStateHandle[ReaderIntent.EXTRA_BRANCH]
   646→   646→        // continue reading
   647→   647→        val history = historyRepository.getOne(manga)
   648→   648→        if (history != null) {
   649→   649→            val chapter = manga.findChapterById(history.chapterId) ?: return null
   650→   650→            // specified branch is requested
   651→   651→            return if (ReaderIntent.EXTRA_BRANCH in savedStateHandle) {
   652→   652→                if (chapter.branch == requestedBranch) {
   653→   653→                    ReaderState(history)
   654→   654→                } else {
   655→   655→                    ReaderState(manga, requestedBranch)
   656→   656→                }
   657→   657→            } else {
   658→   658→                ReaderState(history)
   659→   659→            }
   660→   660→        }
   661→   661→
   662→   662→        // start from beginning
   663→   663→        val preferredBranch = requestedBranch ?: manga.getPreferredBranch(null)
   664→   664→        return ReaderState(manga, preferredBranch)
   665→   665→    }
