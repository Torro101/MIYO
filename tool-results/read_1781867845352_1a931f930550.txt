   309→    fun switchChapterBy(delta: Int) {
   310→        val prevJob = loadingJob
   311→        loadingJob = launchLoadingJob(Dispatchers.Default) {
   312→            prevJob?.cancelAndJoin()
   313→            val prevState = readingState.requireValue()
   314→            val newChapterId = if (delta != 0) {
   315→                val allChapters = mangaDetails.requireValue().allChapters
   316→                var index = allChapters.indexOfFirst { x -> x.id == prevState.chapterId }
   317→                if (index < 0) {
   318→                    return@launchLoadingJob
   319→                }
   320→                index += delta
   321→                (allChapters.getOrNull(index) ?: return@launchLoadingJob).id
   322→            } else {
   323→                prevState.chapterId
   324→            }
   325→            content.value = ReaderContent(emptyList(), null)
   326→            chaptersLoader.loadSingleChapter(newChapterId)
   327→            val newState = ReaderState(
   328→                chapterId = newChapterId,
   329→                page = if (delta == 0) prevState.page else 0,
   330→                scroll = if (delta == 0) prevState.scroll else 0,
   331→            )
   332→            preloadNextChapterAfterLocal(newState.chapterId, mangaDetails.requireValue())
   333→            content.value = ReaderContent(chaptersLoader.snapshot(), newState)
   334→            saveCurrentState(newState)
   335→        }
   336→    }

   338→    @MainThread
   339→    fun onCurrentPageChanged(lowerPos: Int, upperPos: Int) {
   340→        val prevJob = stateChangeJob
   341→        val pages = content.value.pages // capture immediately
   342→        val centerPos = (lowerPos + upperPos) / 2
   343→        val capturedPage = pages.getOrNull(centerPos)
   344→        // Capture the visible window relative to the captured center, so the
   345→        // relative position around the page is preserved even when the page
   346→        // list changes (chapter switch) between the main-thread capture and
   347→        // the post-join recompute below.
   348→        val visibleBeforeCapture = (centerPos - lowerPos).coerceAtLeast(0)
   349→        val visibleAfterCapture = (upperPos - centerPos).coerceAtLeast(0)
   350→        stateChangeJob = launchJob(Dispatchers.Default) {
   351→            prevJob?.cancelAndJoin()
   352→            loadingJob?.join()
   353→            val currentPages = content.value.pages
   354→            val resolvedCenterPos = capturedPage?.let { page ->
   355→                currentPages.indexOfFirst {
   356→                    it.id == page.id && it.chapterId == page.chapterId && it.index == page.index
   357→                }.takeIf { it >= 0 }
   358→            } ?: centerPos.coerceIn(0, currentPages.lastIndex.coerceAtLeast(0))
   359→            val currentPage = currentPages.getOrNull(resolvedCenterPos)
   360→            if (currentPage != null) {
   361→                readingState.update { cs ->
   362→                    cs?.copy(chapterId = currentPage.chapterId, page = currentPage.index)
   363→                }
   364→            }
   365→            notifyStateChanged()
   366→            if (currentPages.isEmpty() || loadingJob?.isActive == true) {
   367→                return@launchJob
   368→            }
   369→            ensureActive()
   370→            // Clamp the captured window to the post-join page list size so we
   371→            // never try to read or prefetch past the end after a chapter shrink.
   372→            val lastIndex = currentPages.lastIndex
   373→            val visibleBefore = visibleBeforeCapture.coerceAtMost(resolvedCenterPos)
   374→            val visibleAfter = visibleAfterCapture.coerceAtMost(lastIndex - resolvedCenterPos)
   375→            val resolvedLowerPos = (resolvedCenterPos - visibleBefore).coerceAtLeast(0)
   376→            val resolvedUpperPos = (resolvedCenterPos + visibleAfter).coerceAtMost(lastIndex)
   377→            val autoLoadAllowed = readerMode.value != ReaderMode.WEBTOON || !isWebtoonPullGestureEnabled.value
   378→            if (autoLoadAllowed) {
   379→                if (resolvedUpperPos >= currentPages.lastIndex - BOUNDS_PAGE_OFFSET) {
   380→                    loadPrevNextChapter(currentPages.last().chapterId, isNext = true)
   381→                }
   382→                if (resolvedLowerPos <= BOUNDS_PAGE_OFFSET) {
   383→                    loadPrevNextChapter(currentPages.first().chapterId, isNext = false)
   384→                }
   385→            }
   386→            val prefetchPages = currentPages.trySublist(
   387→                fromIndex = resolvedUpperPos + 1,
   388→                toIndex = resolvedUpperPos + 1 + PREFETCH_LIMIT,
   389→            )
   390→            if (pageLoader.isPrefetchApplicable(prefetchPages)) {
   391→                pageLoader.prefetch(prefetchPages)
   392→            }
   393→        }
   394→    }

   396→    fun toggleBookmark() {
   397→        if (bookmarkJob?.isActive == true) {
   398→            return
   399→        }
   400→        bookmarkJob = launchJob(Dispatchers.Default) {
   401→            loadingJob?.join()

   405→                bookmarksRepository.removeBookmark(manga.id, state.chapterId, state.page)
   406→                onShowToast.call(R.string.bookmark_removed)
   407→            } else {
   408→                val page = checkNotNull(getCurrentPage()) { "Page not found" }
   409→                val bookmark = Bookmark(
   410→                    manga = requireManga(),
   411→                    pageId = page.id,
   412→                    chapterId = state.chapterId,
   413→                    page = state.page,
   414→                    scroll = state.scroll,
   415→                    imageUrl = page.preview.ifNullOrEmpty { page.url },
   416→                    createdAt = Instant.now(),
   417→                    percent = computePercent(state.chapterId, state.page),
   418→                )
   419→                bookmarksRepository.addBookmark(bookmark)
   420→                onShowToast.call(R.string.bookmark_added)
   421→            }
   422→        }
   423→    }

   432→    private fun loadImpl() {
   433→        loadingJob = launchLoadingJob(Dispatchers.Default + EventExceptionHandler(onLoadingError)) {
   434→            var exception: Exception? = null
   435→            var loadedDetails: MangaDetails? = null
   436→            try {
   437→                detailsLoadUseCase(intent, force = false, preferLocalChapters = true)
   438→                    .collect { details ->
   439→                        loadedDetails = details
   440→                        if (mangaDetails.value == null) {
   441→                            mangaDetails.value = details
   442→                        }
   443→                        chaptersLoader.init(details)
   444→                        val manga = details.toManga()
   445→                        // obtain state
   446→                        if (readingState.value == null) {
   447→                            val newState = getStateFromIntent(manga)
   448→                            if (newState == null) {
   449→                                return@collect // manga not loaded yet if cannot get state
   450→                            }
   451→                            if (!details.canLoadInitialChapter(newState.chapterId)) {
   452→                                mangaDetails.value = details.filterChapters(selectedBranch.value)
   453→                                return@collect
   454→                            }
   455→                            readingState.value = newState
   456→                            val mode = runCatchingCancellable {
   457→                                detectReaderModeUseCase(manga, newState)
   458→                            }.getOrDefault(settings.defaultReaderMode)
   459→                            val branch = chaptersLoader.peekChapter(newState.chapterId)?.branch
   460→                            selectedBranch.value = branch
   461→                            readerMode.value = mode
   462→                            try {
   463→                                chaptersLoader.loadSingleChapter(newState.chapterId)
   464→                                preloadNextChapterAfterLocal(newState.chapterId, details)
   465→                            } catch (e: Exception) {
   466→                                readingState.value = null // try next time
   467→                                exception = e.mergeWith(exception)
   468→                                return@collect
   469→                            }
   470→                        }
   471→                        mangaDetails.value = details.filterChapters(selectedBranch.value)
   472→
   473→                        // save state
   474→                        if (!isIncognitoMode.firstNotNull()) {
   475→                            readingState.value?.let {
   476→                                val percent = computePercent(it.chapterId, it.page)
   477→                                historyUpdateUseCase(manga, it, percent)
   478→                            }
   479→                        }
   480→                        notifyStateChanged()
   481→                        content.value = ReaderContent(chaptersLoader.snapshot(), readingState.value)
   482→                    }
   483→            } catch (e: CancellationException) {
   484→                throw e
   485→            } catch (e: Exception) {
   486→                exception = e.mergeWith(exception)
   487→            }
   488→            if (readingState.value == null) {
   489→                val loadedManga = loadedDetails // for smart cast
   490→                if (loadedManga != null) {
   491→                    mangaDetails.value = loadedManga.filterChapters(selectedBranch.value)
   492→                }
   493→                val loadingError = when {
   494→                    exception != null -> exception
   495→                    loadedManga == null || !loadedManga.isLoaded -> null
   496→                    loadedManga.isRestricted -> EmptyMangaException(
   497→                        EmptyMangaReason.RESTRICTED,
   498→                        loadedManga.toManga(),
   499→                        null,
   500→                    )
   501→
   502→                    loadedManga.allChapters.isEmpty() -> EmptyMangaException(
   503→                        EmptyMangaReason.NO_CHAPTERS,
   504→                        loadedManga.toManga(),
   505→                        null,
   506→                    )
   507→
   508→                    else -> null
   509→                } ?: IllegalStateException("Unable to load manga. This should never happen. Please report")
   510→                onLoadingError.call(loadingError)
   511→            } else exception?.let { e ->
   512→                // manga has been loaded but error occurred
   513→                errorEvent.call(e)
   514→            }
   515→        }
   516→    }

   518→    private fun MangaDetails.canLoadInitialChapter(chapterId: Long): Boolean {

   522→    private suspend fun preloadNextChapterAfterLocal(currentId: Long, details: MangaDetails) {
   523→        if (!settings.isPagesPreloadEnabled) {
   524→            return
   525→        }
   526→        if (readerMode.value == ReaderMode.WEBTOON && isWebtoonPullGestureEnabled.value) {
   527→            return
   528→        }
   529→        val chapter = chaptersLoader.peekChapter(currentId) ?: return
   530→        if (chapter.source != LocalMangaSource) {
   531→            return
   532→        }
   533→        runCatchingCancellable {
   534→            chaptersLoader.loadPrevNextChapter(details, currentId, isNext = true)
   535→        }.getOrNull()
   536→    }

   548→    private fun <T> List<T>.trySublist(fromIndex: Int, toIndex: Int): List<T> {
   549→        val fromIndexBounded = fromIndex.coerceIn(0, size)
   550→        val toIndexBounded = toIndex.coerceIn(fromIndexBounded, size)
   551→        return if (fromIndexBounded == toIndexBounded) {
   552→            emptyList()
   553→        } else {
   554→            subList(fromIndexBounded, toIndexBounded)
   555→        }
   556→    }

   558→    @WorkerThread
   559→    private fun notifyStateChanged() {
   560→        val state = getCurrentState() ?: return
   561→        val chapter = chaptersLoader.peekChapter(state.chapterId) ?: return
   562→        val m = mangaDetails.value ?: return
   563→        val chapterIndex = m.chapters[chapter.branch]?.indexOfFirst { it.id == chapter.id } ?: -1
   564→        val newState = ReaderUiState(
   565→            mangaName = m.toManga().title,
   566→            chapter = chapter,
   567→            chapterIndex = chapterIndex,
   568→            chaptersTotal = m.chapters[chapter.branch].sizeOrZero(),
   569→            totalPages = chaptersLoader.getPagesCount(chapter.id),
   570→            currentPage = state.page,
   571→            percent = computePercent(state.chapterId, state.page),
   572→            incognito = isIncognitoMode.value == true,
   573→        )
   574→        uiState.value = newState
   575→        if (isIncognitoMode.value == false) {
   576→            statsCollector.onStateChanged(m.id, state)
   577→            discordRpc.updateRpc(m.toManga(), newState)
   578→        }
   579→    }

   581→    private fun computePercent(chapterId: Long, pageIndex: Int): Float {
   582→        val branch = chaptersLoader.peekChapter(chapterId)?.branch
   583→        val chapters = mangaDetails.value?.chapters?.get(branch) ?: return PROGRESS_NONE
   584→        val chaptersCount = chapters.size
   585→        val chapterIndex = chapters.indexOfFirst { x -> x.id == chapterId }
   586→        val pagesCount = chaptersLoader.getPagesCount(chapterId)
   587→        if (chaptersCount == 0 || pagesCount == 0) {
   588→            return PROGRESS_NONE
   589→        }
   590→        val pagePercent = (pageIndex + 1) / pagesCount.toFloat()
   591→        val ppc = 1f / chaptersCount
   592→        return ppc * chapterIndex + ppc * pagePercent
   593→    }

   630→    private suspend fun getStateFromIntent(manga: Manga): ReaderState? {
   631→        // check if we have at least some chapters loaded
   632→        if (manga.chapters.isNullOrEmpty()) {
   633→            return null
   634→        }
   635→        // specific state is requested
   636→        val requestedState: ReaderState? = savedStateHandle[ReaderIntent.EXTRA_STATE]
   637→        if (requestedState != null) {
   638→            return if (manga.findChapterById(requestedState.chapterId) != null) {
   639→                requestedState
   640→            } else {
   641→                null
   642→            }
   643→        }
   644→
   645→        val requestedBranch: String? = savedStateHandle[ReaderIntent.EXTRA_BRANCH]
   646→        // continue reading
   647→        val history = historyRepository.getOne(manga)
   648→        if (history != null) {
   649→            val chapter = manga.findChapterById(history.chapterId) ?: return null
   650→            // specified branch is requested
   651→            return if (ReaderIntent.EXTRA_BRANCH in savedStateHandle) {
   652→                if (chapter.branch == requestedBranch) {
   653→                    ReaderState(history)
   654→                } else {
   655→                    ReaderState(manga, requestedBranch)
   656→                }
   657→            } else {
   658→                ReaderState(history)
   659→            }
   660→        }
   661→
   662→        // start from beginning
   663→        val preferredBranch = requestedBranch ?: manga.getPreferredBranch(null)
   664→        return ReaderState(manga, preferredBranch)
   665→    }
