   121→   121→        fun isPrefetchApplicable(pages: List<ReaderPage>): Boolean {
   122→   122→                val repo = repository
   123→   123→                val hasLocalPages = repo?.source == LocalMangaSource || pages.any { it.isLocalPage() }

   125→   125→                return canPrefetchSource

   131→   131→        @AnyThread
   132→   132→        fun prefetch(pages: List<ReaderPage>) = loaderScope.launch {
   133→   133→                var hasPendingPrefetch = false
   134→   134→                prefetchLock.withLock {
   135→   135→                        val queueLimit = getPrefetchQueueLimit()
   136→   136→                        if (queueLimit <= 0) {
   137→   137→                                prefetchQueue.clear()
   138→   138→                                return@withLock
   139→   139→                        }
   140→   140→                        for (page in pages.asReversed()) {
   141→   141→                                val key = page.toMangaPage().taskKey()
   142→   142→                                val hasTask = synchronized(tasks) {
   143→   143→                                        tasks.containsKey(key)
   144→   144→                                }
   145→   145→                                if (hasTask) {
   146→   146→                                        continue
   147→   147→                                }
   148→   148→                                prefetchQueue.offerFirst(page.toMangaPage())
   149→   149→                                if (prefetchQueue.size > queueLimit) {
   150→   150→                                        prefetchQueue.pollLast()
   151→   151→                                }
   152→   152→                        }
   153→   153→                        hasPendingPrefetch = prefetchQueue.isNotEmpty()
   154→   154→                }
   155→   155→                if (hasPendingPrefetch && counter.get() == 0) {
   156→   156→                        onIdle()
   157→   157→                }
   158→   158→        }

   217→   217→        suspend fun materializeZipEntry(uri: Uri): Uri {
   218→   218→                check(uri.isZipUri()) { "Expected ZIP uri: $uri" }
   219→   219→                val cacheKey = rawZipCacheKey(uri)
   220→   220→                cache.get(cacheKey)?.let { file ->
   221→   221→                        if (file.isUsablePageCache()) {
   222→   222→                                return file.toUri()
   223→   223→                        }
   224→   224→                        cache.remove(cacheKey)
   225→   225→                }
   226→   226→                return withContext(Dispatchers.IO) {
   227→   227→                        ZipFile(uri.schemeSpecificPart).use { zip ->
   228→   228→                                val entryName = uri.fragment ?: throw FileNotFoundException(uri.toString())
   229→   229→                                val entry = zip.getEntry(entryName) ?: throw FileNotFoundException("$uri#$entryName")
   230→   230→                                zip.getInputStream(entry).source().use { source ->
   231→   231→                                        val file = cache.set(cacheKey, source, MimeTypes.getMimeTypeFromExtension(entry.name))
   232→   232→                                        if (file.isUsablePageCache()) {
   233→   233→                                                file.toUri()
   234→   234→                                        } else {
   235→   235→                                                cache.remove(cacheKey)
   236→   236→                                                throw UnsupportedFileException("Unsupported or corrupt image file")
   237→   237→                                        }
   238→   238→                                }
   239→   239→                        }
   240→   240→                }
   241→   241→        }

   243→   243→        @CheckResult
   244→   244→        suspend fun convertBimap(uri: Uri): Uri = convertLock.withLock {
   245→   245→                if (uri.isZipUri()) {
   246→   246→                        val cacheKey = convertedZipCacheKey(uri)
   247→   247→                        cache.get(cacheKey)?.let { file ->
   248→   248→                                if (file.isUsablePageCache()) {
   249→   249→                                        return@withLock file.toUri()
   250→   250→                                }
   251→   251→                                cache.remove(cacheKey)
   252→   252→                        }
   253→   253→                        runInterruptible(Dispatchers.IO) {
   254→   254→                                ZipFile(uri.schemeSpecificPart).use { zip ->
   255→   255→                                        val entryName = uri.fragment ?: throw FileNotFoundException(uri.toString())
   256→   256→                                        val entry = zip.getEntry(entryName) ?: throw FileNotFoundException("$uri#$entryName")
   257→   257→                                        context.ensureRamAtLeast(estimateDecodeBytes(entry.size.coerceAtLeast(0L)))
   258→   258→                                        zip.getInputStream(entry).use {
   259→   259→                                                BitmapDecoderCompat.decode(it, MimeTypes.getMimeTypeFromExtension(entry.name))
   260→   260→                                        }
   261→   261→                                }
   262→   262→                        }.use { image ->
   263→   263→                                cache.set(cacheKey, image).toUri()
   264→   264→                        }
   265→   265→                } else {
   266→   266→                        // For a plain file URI, write the converted PNG to a NEW cache
   267→   267→                        // entry (not back to the source) so the original is preserved and
   268→   268→                        // the conversion works even when the source lives in read-only
   269→   269→                        // storage (e.g. SAF tree the user only granted read access to).
   270→   270→                        val cacheKey = convertedFileCacheKey(uri)
   271→   271→                        cache.get(cacheKey)?.let { file ->
   272→   272→                                if (file.isUsablePageCache()) {
   273→   273→                                        return@withLock file.toUri()
   274→   274→                                }
   275→   275→                                cache.remove(cacheKey)
   276→   276→                        }
   277→   277→                        val source = uri.toFile()
   278→   278→                        runInterruptible(Dispatchers.IO) {
   279→   279→                                context.ensureRamAtLeast(source.estimateDecodeMemoryBytes())
   280→   280→                                BitmapDecoderCompat.decode(source)
   281→   281→                        }.use { image ->
   282→   282→                                cache.set(cacheKey, image).toUri()
   283→   283→                        }
   284→   284→                }
   285→   285→        }

   315→   315→        private fun onIdle() = loaderScope.launch {
   316→   316→                prefetchLock.withLock {
   317→   317→                        while (prefetchQueue.isNotEmpty()) {
   318→   318→                                val page = prefetchQueue.pollFirst() ?: return@launch

   326→   326→        private fun loadPageAsyncImpl(
   327→   327→                page: MangaPage,
   328→   328→                skipCache: Boolean,
   329→   329→                isPrefetch: Boolean,
   330→   330→        ): ProgressDeferred<Uri, Float> {
   331→   331→                val progress = MutableStateFlow(PROGRESS_UNDEFINED)
   332→   332→                val deferred = loaderScope.async {
   333→   333→                        counter.incrementAndGet()
   334→   334→                        try {
   335→   335→                                loadPageImpl(
   336→   336→                                        page = page,
   337→   337→                                        progress = progress,
   338→   338→                                        isPrefetch = isPrefetch,
   339→   339→                                        skipCache = skipCache,
   340→   340→                                )
   341→   341→                        } finally {
   342→   342→                                if (counter.decrementAndGet() == 0) {
   343→   343→                                        onIdle()
   344→   344→                                }
   345→   345→                        }
   346→   346→                }
   347→   347→                return ProgressDeferred(deferred, progress)
   348→   348→        }

   360→   360→        private suspend fun loadPageImpl(
   361→   361→                page: MangaPage,
   362→   362→                progress: MutableStateFlow<Float>,
   363→   363→                isPrefetch: Boolean,
   364→   364→                skipCache: Boolean,
   365→   365→        ): Uri = semaphore.withPermit {
   366→   366→                val pageUrl = resolvePageUrl(page)
   367→   367→                check(pageUrl.isNotBlank()) { "Cannot obtain full image url for $page" }
   368→   368→                if (!skipCache) {
   369→   369→                        cache.get(pageUrl)?.let { file ->
   370→   370→                                if (file.isUsablePageCache()) {
   371→   371→                                        recordLoadedPageFootprint(file)
   372→   372→                                        return file.toUri()
   373→   373→                                }
   374→   374→                                cache.remove(pageUrl)
   375→   375→                        }
   376→   376→                }
   377→   377→                val uri = pageUrl.toUri()
   378→   378→                return when {
   379→   379→                        uri.isZipUri() -> {
   380→   380→                                val normalizedUri = if (uri.scheme == URI_SCHEME_ZIP) {
   381→   381→                                        uri
   382→   382→                                } else { // legacy uri
   383→   383→                                        uri.buildUpon().scheme(URI_SCHEME_ZIP).build()
   384→   384→                                }
   385→   385→                                materializeZipEntry(normalizedUri).also {
   386→   386→                                        recordLoadedPageFootprint(it.toFile())
   387→   387→                                }
   388→   388→                        }
   389→   389→
   390→   390→                        uri.isFileUri() -> uri.also {
   391→   391→                                if (!it.toFile().isUsablePageCache()) {
   392→   392→                                        throw UnsupportedFileException("Unsupported or corrupt image file")
   393→   393→                                }
   394→   394→                                recordLoadedPageFootprint(it.toFile())
   395→   395→                        }
   396→   396→                        else -> {
   397→   397→                                if (isPrefetch) {
   398→   398→                                        val slowdownDelayMs = downloadSlowdownDispatcher.getDelayMs(page.source)
   399→   399→                                        if (slowdownDelayMs > 0L) {
   400→   400→                                                delay(slowdownDelayMs)
   401→   401→                                        }
   402→   402→                                }
   403→   403→                                val request = createPageRequest(pageUrl, page.source)
   404→   404→                                imageProxyInterceptor.interceptPageRequest(request, okHttp).ensureSuccess().use { response ->
   405→   405→                                        response.requireBody().withProgress(progress).use {
   406→   406→                                                cache.set(pageUrl, it.source(), it.contentType()?.toMimeType())
   407→   407→                                        }
   408→   408→                                }.also { file ->
   409→   409→                                        if (!file.isUsablePageCache()) {
   410→   410→                                                cache.remove(pageUrl)
   411→   411→                                                throw UnsupportedFileException("Unsupported or corrupt image file")
   412→   412→                                        }
   413→   413→                                        recordLoadedPageFootprint(file)
   414→   414→                                }.toUri()
   415→   415→                        }
   416→   416→                }
   417→   417→        }

   437→   437→        private fun getPrefetchQueueLimit(): Int {
   438→   438→                val availableRam = context.ramAvailable
   439→   439→                if (context.isPowerSaveMode() || isLowRam(availableRam)) {
   440→   440→                        return 0
   441→   441→                }
   442→   442→                val baseLimit = when {
   443→   443→                        availableRam >= FileSize.MEGABYTES.convert(PREFETCH_HIGH_RAM_MB, FileSize.BYTES) -> PREFETCH_LIMIT_HIGH
   444→   444→                        availableRam >= FileSize.MEGABYTES.convert(PREFETCH_DEFAULT_RAM_MB, FileSize.BYTES) -> PREFETCH_LIMIT_DEFAULT
   445→   445→                        else -> PREFETCH_LIMIT_LOW
   446→   446→                }
   447→   447→                return when {
   448→   448→                        lastPageDecodeBytes >= megabytes(PREFETCH_HUGE_PAGE_MB) -> minOf(baseLimit, PREFETCH_LIMIT_HUGE_PAGE)
   449→   449→                        lastPageDecodeBytes >= megabytes(PREFETCH_LARGE_PAGE_MB) -> minOf(baseLimit, PREFETCH_LIMIT_LARGE_PAGE)
   450→   450→                        else -> baseLimit
   451→   451→                }
   452→   452→        }

   494→   494→        private fun estimateDecodeBytes(encodedBytes: Long): Long {
   495→   495→                val normalizedBytes = encodedBytes.coerceAtLeast(0L)
   496→   496→                val scaledBytes = if (normalizedBytes > Long.MAX_VALUE / IMAGE_DECODE_FALLBACK_MULTIPLIER) {
   497→   497→                        Long.MAX_VALUE
   498→   498→                } else {
   499→   499→                        normalizedBytes * IMAGE_DECODE_FALLBACK_MULTIPLIER
   500→   500→                }
   501→   501→                return scaledBytes
   502→   502→                        .coerceAtLeast(megabytes(IMAGE_DECODE_MIN_MB))
   503→   503→                        .coerceAtMost(megabytes(IMAGE_DECODE_MAX_MB))
   504→   504→        }
